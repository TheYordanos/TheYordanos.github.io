Flocking Simulation: The Boids Algorithm

March 19, 2026

In 1986, some computer graphics researcher called Craig Reynolds wanted to show birds on the big glassy screen. He introduced a simple yet elegant algorithm to simulate the flocking behavior of birds. He called his simulated creatures "boids". A clipped version of "bird-oids". What makes this algorithm facinating to me is the complex group behavior that emerges from just three simple rules.

The Three Fundamental Rules

  • Separation: Each boid steers to avoid crowding its neighbors.
  • Alignment: Each boid steers toward the average heading of nearby boids. This causes the flock to move together.
  • Cohesion: Each boid steers toward the average position of nearby boids. This keeps the flock together.

Thats it... Thats the whole simulation. Complex group behavior emerging from simple individual interactions :)

Some additional behaviours I added

This implementation adds two more behaviors to give the simulation more... *oomph*:

  • Wall avoidance: Boids sense and steer away from the canvas edges.
  • Speed control: Boids maintain a minimum and maximum speed.

MATH

wait don't go away because of this yet... its actually really simple.

Each boid has a position and velocity vector. On every frame, every boid looks at its neighbors within a vision radius. For each of the three rules, it calculates a desired steering force, then combines them using weighted averages. These weights determine the personality of the flock:

  • High separation weight = scattered, cautious flock
  • High alignment weight = tightly coordinated, school-like movement
  • High cohesion weight = dense, clustered flock

Details

The simulation runs at 60 frames per second using requestAnimationFrame. Each frame:

  1. Calculate new velocities for all boids based on neighbors
  2. Apply speed limits to keep motion natural
  3. Update positions
  4. Wrap around screen edges for infinite space
  5. Render each boid as a colored triangle pointing in its direction

The color of each boid is determined by its direction (angle), it creates a beautiful rainbow effect as the flock swirls and turns. I like it :)

The Code

Here are parts of the complete implementation. The magic happens in the separation, alignment, and cohesion functions:

CONSTANTS

const MAX_VEL = 8
const MIN_VEL = 3
const MAX_FORCE = 0.6
const ALIGNMENT_WEIGHT = 0.6
const COHESION_WEIGHT = 0.4
const SEPARATION_WEIGHT = 2.4
const WALL_AVOID_WEIGHT = 1.6

const VISION_DISTANCE = 60
const SEPARATION_DISTANCE = 25
const WALL_MARGINS = 40

INITIAL FUNCTION

This function runs at the beginning of the simulation. It gives each boid a random position and velocity.

function init() {

  tick()

TICK

This function runs 60 frames per second and is the main loop of the sim. It handles updating and drawing.

function tick() {
  let newVelocities = []

  boids.forEach((boid, i) => {
    let sep = separation(i)
    let ali = alignment(i)
    let coh = cohesion(i)
    let wall = avoid_walls(i)

    let totalForce = {
      x: sep.x * SEPARATION_WEIGHT + ali.x * ALIGNMENT_WEIGHT + coh.x * COHESION_WEIGHT + wall.x * WALL_AVOID_WEIGHT,
      y: sep.y * SEPARATION_WEIGHT + ali.y * ALIGNMENT_WEIGHT + coh.y * COHESION_WEIGHT + wall.y * WALL_AVOID_WEIGHT,
    }

    totalForce = limit_vec(totalForce, MAX_FORCE)

    let newVel = {
      x: boid.vel.x + totalForce.x,
      y: boid.vel.y + totalForce.y,
    }

    let speed = Math.sqrt(newVel.x * newVel.x + newVel.y * newVel.y)
    if (speed > MAX_VEL) {
      newVel.x = (newVel.x / speed) * MAX_VEL
      newVel.y = (newVel.y / speed) * MAX_VEL
    } else if (speed < MIN_VEL && speed > 0) {
      newVel.x = (newVel.x / speed) * MIN_VEL
      newVel.y = (newVel.y / speed) * MIN_VEL
    }

    newVelocities[i] = newVel
  })

  boids.forEach((boid, i) => {
    boid.vel.x = newVelocities[i].x
    boid.vel.y = newVelocities[i].y

    boid.pos.x += boid.vel.x
    boid.pos.y += boid.vel.y

    if (boid.pos.x < 0) boid.pos.x = canvas.width
    if (boid.pos.x > canvas.width) boid.pos.x = 0
    if (boid.pos.y < 0) boid.pos.y = canvas.height
    if (boid.pos.y > canvas.height) boid.pos.y = 0
  })

  ctx.clearRect(0, 0, canvas.width, canvas.height)

  ctx.strokeStyle = "#222"
  ctx.lineWidth = 1
  for (let i = 0; i < canvas.width; i += 50) {
    ctx.beginPath()
    ctx.moveTo(i, 0)
    ctx.lineTo(i, canvas.height)
    ctx.stroke()
  }
  for (let i = 0; i < canvas.height; i += 50) {
    ctx.beginPath()
    ctx.moveTo(0, i)
    ctx.lineTo(canvas.width, i)
    ctx.stroke()
  }

  boids.forEach((boid) => {
    drawTriangleBoid(boid)
  })

  requestAnimationFrame(tick)
}

function drawTriangleBoid(boid) {
  let angle = Math.atan2(boid.vel.y, boid.vel.x)
  let size = boid.size

  let tip = {
    x: boid.pos.x + Math.cos(angle) * size * 1.5,
    y: boid.pos.y + Math.sin(angle) * size * 1.5,
  }

  let left = {
    x: boid.pos.x + Math.cos(angle + 2.2) * size,
    y: boid.pos.y + Math.sin(angle + 2.2) * size,
  }

  let right = {
    x: boid.pos.x + Math.cos(angle - 2.2) * size,
    y: boid.pos.y + Math.sin(angle - 2.2) * size,
  }

  ctx.beginPath()
  ctx.moveTo(tip.x, tip.y)
  ctx.lineTo(left.x, left.y)
  ctx.lineTo(right.x, right.y)
  ctx.closePath()

  let hue = (angle * 180 / Math.PI + 180) % 360
  let gradient = ctx.createLinearGradient(boid.pos.x - size, boid.pos.y - size, boid.pos.x + size, boid.pos.y + size)
  gradient.addColorStop(0, `hsl(${hue}, 80%, 70%)`)
  gradient.addColorStop(1, `hsl(${hue}, 80%, 50%)`)

  ctx.fillStyle = gradient
  ctx.fill()
  ctx.strokeStyle = "rgba(255,255,255,0.3)"
  ctx.lineWidth = 1
  ctx.stroke()

  ctx.beginPath()
  ctx.arc(boid.pos.x + Math.cos(angle) * size * 0.5, boid.pos.y + Math.sin(angle) * size * 0.5, size * 0.2, 0, 2 * Math.PI)
  ctx.fillStyle = "white"
  ctx.fill()
  ctx.strokeStyle = "black"
  ctx.lineWidth = 1
  ctx.stroke()
}

UTITLITY FUNCTIONS

function normalize(vec) {
  let mag = Math.sqrt(vec.x * vec.x + vec.y * vec.y)
  if (mag < 0.0001) return { x: 0, y: 0 }
  return { x: vec.x / mag, y: vec.y / mag }
}

function distance(vec1, vec2) {
  let dx = vec2.x - vec1.x
  let dy = vec2.y - vec1.y
  return Math.sqrt(dx * dx + dy * dy)
}

function limit_vec(vec, max) {
  let mag = Math.sqrt(vec.x * vec.x + vec.y * vec.y)
  if (mag > max && mag > 0) {
    return { x: (vec.x / mag) * max, y: (vec.y / mag) * max }
  }
  return { x: vec.x, y: vec.y }
}

HEART OF THE SIMULATION

Separation

function separation(idx) {
  let curr = boids[idx]
  let steer = { x: 0, y: 0 }
  let count = 0

  boids.forEach((other, i) => {
    if (idx == i) return

    let dist = distance({ x: curr.pos.x, y: curr.pos.y }, { x: other.pos.x, y: other.pos.y })

    if (dist > 0 && dist < SEPARATION_DISTANCE) {
      let diff = {
        x: curr.pos.x - other.pos.x,
        y: curr.pos.y - other.pos.y,
      }
      diff = normalize(diff)
      diff.x /= dist
      diff.y /= dist

      steer.x += diff.x
      steer.y += diff.y
      count++
    }
  })

  if (count > 0) {
    steer.x /= count
    steer.y /= count
  }

  if (Math.sqrt(steer.x * steer.x + steer.y * steer.y) > 0) {
    steer = normalize(steer)
    steer.x *= MAX_VEL
    steer.y *= MAX_VEL
    steer.x -= curr.vel.x
    steer.y -= curr.vel.y
    steer = limit_vec(steer, MAX_FORCE)
  }

  return steer
}

Alignment

function alignment(idx) {
  let curr = boids[idx]
  let avgVel = { x: 0, y: 0 }
  let count = 0

  boids.forEach((other, i) => {
    if (idx == i) return

    let dist = distance({ x: curr.pos.x, y: curr.pos.y }, { x: other.pos.x, y: other.pos.y })

    if (dist < VISION_DISTANCE) {
      avgVel.x += other.vel.x
      avgVel.y += other.vel.y
      count++
    }
  })

  if (count > 0) {
    avgVel.x /= count
    avgVel.y /= count

    avgVel = normalize(avgVel)
    avgVel.x *= MAX_VEL
    avgVel.y *= MAX_VEL

    let steer = {
      x: avgVel.x - curr.vel.x,
      y: avgVel.y - curr.vel.y,
    }
    return limit_vec(steer, MAX_FORCE)
  }

  return { x: 0, y: 0 }
}

Cohesion

function cohesion(idx) {
  let curr = boids[idx]
  let center = { x: 0, y: 0 }
  let count = 0

  boids.forEach((other, i) => {
    if (idx == i) return

    let dist = distance({ x: curr.pos.x, y: curr.pos.y }, { x: other.pos.x, y: other.pos.y })

    if (dist < VISION_DISTANCE) {
      center.x += other.pos.x
      center.y += other.pos.y
      count++
    }
  })

  if (count > 0) {
    center.x /= count
    center.y /= count

    let desired = {
      x: center.x - curr.pos.x,
      y: center.y - curr.pos.y,
    }

    let distToCenter = Math.sqrt(desired.x * desired.x + desired.y * desired.y)
    if (distToCenter > 0) {
      desired.x /= distToCenter
      desired.y /= distToCenter
      desired.x *= MAX_VEL
      desired.y *= MAX_VEL

      let steer = {
        x: desired.x - curr.vel.x,
        y: desired.y - curr.vel.y,
      }
      return limit_vec(steer, MAX_FORCE)
    }
  }

  return { x: 0, y: 0 }
}

Wall Avoidance

function avoid_walls(idx) {
  let curr = boids[idx]
  let steer = { x: 0, y: 0 }

  if (curr.pos.x < WALL_MARGINS) {
    steer.x += (WALL_MARGINS - curr.pos.x) / WALL_MARGINS
  }
  if (curr.pos.x > canvas.width - WALL_MARGINS) {
    steer.x -= (curr.pos.x - (canvas.width - WALL_MARGINS)) / WALL_MARGINS
  }
  if (curr.pos.y < WALL_MARGINS) {
    steer.y += (WALL_MARGINS - curr.pos.y) / WALL_MARGINS
  }
  if (curr.pos.y > canvas.height - WALL_MARGINS) {
    steer.y -= (curr.pos.y - (canvas.height - WALL_MARGINS)) / WALL_MARGINS
  }

  if (Math.sqrt(steer.x * steer.x + steer.y * steer.y) > 0) {
    steer = normalize(steer)
    steer.x *= MAX_VEL
    steer.y *= MAX_VEL
    steer.x -= curr.vel.x
    steer.y -= curr.vel.y
    steer = limit_vec(steer, MAX_FORCE * 2)
  }

  return steer
}
